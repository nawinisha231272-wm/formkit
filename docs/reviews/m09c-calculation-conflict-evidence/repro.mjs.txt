import { createEngine } from '@ourform/core';
import { createLogic } from '@ourform/core/logic';
import { textField, numberField } from '@ourform/components';
import { validateSubmission } from '@ourform/validation';
import { writeFileSync } from 'node:fs';
const definition = {$formVersion:'1.0',id:'two-calculations',display:'form',components:[{id:'source',key:'source',type:'text',defaultValue:'yes'},{id:'target',key:'target',type:'number',calculate:{expr:2,allowOverride:false}}],logic:[{id:'conditional-calculation',trigger:{on:'change',refs:['source']},when:['eq',{ref:'source'},'yes'],then:[{action:'calculate',target:'target',expr:3}]}]};
const engine = createEngine({logic:createLogic()}).register(textField.model).register(numberField.model);
const compiled = engine.compile(definition);
if(!compiled.ok)throw Error(JSON.stringify(compiled.diagnostics));
const session=engine.createSession(compiled.form);
const initial=session.getData();
session.setValue('source','no',{source:'user'});
const edited=session.getData();
const client=await session.submit();
if(!client.ok)throw Error(JSON.stringify(client));
const node=await validateSubmission(compiled.form,client.data);
const report={definition,compile:compiled.ok,diagnostics:compiled.diagnostics,initial,edited,client,node};
writeFileSync('.git/ourform-m09c-replay-protected-calc-2026-11-11/results.json',JSON.stringify(report,null,2)+'\n');
console.log(JSON.stringify(report));session.dispose();
