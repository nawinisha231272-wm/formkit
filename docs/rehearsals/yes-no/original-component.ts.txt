import {
  defineField, stringProperty, controlAttributes, h,
} from "@ourform/sdk";

export const field = defineField({
  type: "acme-yes-no",
  title: "Yes/No toggle",
  shell: "inline",
  value: {
    empty: () => null,
    normalize: (value: unknown): boolean | null =>
      value === true ? true : value === false ? false : null,
    serialize: (value: boolean | null) => value,
    equals: (a: boolean | null, b: boolean | null) => a === b,
    isEmpty: (value: boolean | null) => value !== true,
    jsonSchema: () => ({ type: ["boolean", "null"] }),
  },
  properties: {
    yesLabel: stringProperty({ label: "Yes label", default: "Yes" }),
    noLabel: stringProperty({ label: "No label", default: "No" }),
  },
  control: (p) => h("div", {},
    h("input", {
      ...controlAttributes(p),
      type: "checkbox",
      checked: p.value === true,
      class: p.theme.classFor("control"),
      oninput: (event: Event) =>
        p.setValue((event.target as HTMLInputElement).checked),
      onblur: p.onBlur,
    }),
    h("span", {"data-answer-label": ""}, p.t(p.value === true ? p.config.yesLabel : p.config.noLabel)),
  ),
});
